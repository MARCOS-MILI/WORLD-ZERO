rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function logado() { return request.auth != null; }

    // Coloque na lista os e-mails das contas Google dos mestres (tudo em minúsculas, entre aspas simples, separados por vírgula).
    function mestre() {
      return logado()
        && request.auth.token.get('email', '') in ['email.do.mestre1@gmail.com', 'email.do.mestre2@gmail.com']
        && request.auth.token.get('email_verified', false) == true
        && request.auth.token.firebase.sign_in_provider == 'google.com';
    }

    // Qualquer pessoa que entrou (Google ou sem conta) pode ver as fichas e escrever só no próprio documento.
    // O documento "mesa" (fichas do mestre e NPCs revelados) só o mestre escreve.
    match /fichas/{dono} {
      allow read: if logado();
      allow create, update: if mestre() || (logado() && request.auth.uid == dono);
      allow delete: if mestre();
    }

    // NPCs ocultos e anotações: só o mestre lê e escreve.
    match /npcs/{id} { allow read, write: if mestre(); }
    match /gm/{id} { allow read, write: if mestre(); }
  }
}
